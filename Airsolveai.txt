def make_pnr():
    return ''.join(random.choices(
        string.ascii_uppercase + string.digits, k=6
    ))


# ============================================================
# API - CHECK FLIGHT
# ============================================================

@app.route("/api/analyze", methods=["POST"])
def analyze():

    data = request.get_json()

    passenger = data.get("passenger", "").strip()
    flight_number = data.get("flight", "").strip().upper()

    if not passenger:
        return jsonify({
            "success": False,
            "message": "Please enter passenger name."
        }), 400

    if flight_number not in FLIGHTS:
        return jsonify({
            "success": False,
            "message": "Flight not found. Try UK 835, 6E 802, 6E 215, 6E 6143 or AI 540."
        }), 404

    flight = FLIGHTS[flight_number]

    add_notification(
        "Flight Analysis Started",
        f"AirSolveAI started multi-agent analysis for {flight_number}.",
        "info"
    )

    return jsonify({
        "success": True,
        "passenger": passenger,
        "flight_number": flight_number,
        "flight": flight
    })


# ============================================================
# API - ALTERNATIVES
# ============================================================

@app.route("/api/alternatives/<flight_number>")
def alternatives(flight_number):

    flight_number = flight_number.upper()

    flight = FLIGHTS.get(flight_number)

    if not flight:
        return jsonify({
            "success": False,
            "message": "Flight not found."
        }), 404

    if flight["status"] not in ["DELAYED", "CANCELLED"]:
        return jsonify({
            "success": True,
            "available": False,
            "alternatives": []
        })

    result = ALTERNATIVES.get(flight_number, [])

    add_notification(
        "Alternative Flights Found",
        f"{len(result)} recovery flights are available for {flight_number}.",
        "alternative"
    )

    return jsonify({
        "success": True,
        "available": True,
        "alternatives": result
    })

@app.route("/api/seats/<flight_number>")
def seats(flight_number):

    flight_number = flight_number.upper()

    if flight_number not in FLIGHTS:

        # Alternative flight lookup
        found = None

        for group in ALTERNATIVES.values():
            for item in group:
                if item["flight"] == flight_number:
                    found = item
                    break

        if not found:
            return jsonify({
                "success": False
            }), 404

        aircraft = found["aircraft"]

    else:
        aircraft = FLIGHTS[flight_number]["aircraft"]

    occupied = [
        "1A", "1B",
        "2C", "2D",
        "4A", "4F",
        "7B", "7E",
        "10A", "10F",
        "12C", "12D",
        "15A", "15F",
        "18B", "18E",
        "20A", "20F"
    ]

    return jsonify({
        "success": True,
        "aircraft": aircraft,
        "rows": 20,
        "columns": ["A", "B", "C", "D", "E", "F"],
        "occupied": occupied
    })

@app.route("/api/rebook", methods=["POST"])
def rebook():

    data = request.get_json()

    passenger = data.get("passenger", "Passenger")
    old_flight = data.get("old_flight", "")
    new_flight = data.get("new_flight", "")
    seat = data.get("seat", "")

    selected = None

    for group in ALTERNATIVES.values():
        for item in group:
            if item["flight"] == new_flight:
                selected = item
                break

    if not selected:
        return jsonify({
            "success": False,
            "message": "Selected alternative flight not found."
        }), 404

    pnr = make_pnr()

    ticket = {
        "passenger": passenger,
        "old_flight": old_flight,
        "flight": new_flight,
        "airline": selected["airline"],
        "from": selected["from"],
        "to": selected["to"],
        "departure": selected["departure"],
        "arrival": selected["arrival"],
        "date": "10 Oct 2026",
        "seat": seat,
        "pnr": pnr,
        "terminal": selected["terminal"],
        "gate": selected["gate"],
        "aircraft": selected["aircraft"],
        "boarding": "Group B",
        "status": "REBOOKED"
    }

    bookings.insert(0, ticket)

    add_notification(
        "Seat Selected",
        f"Seat {seat} has been selected on {new_flight}.",
        "success"
    )

    add_notification(
        "Rebooking Confirmed",
        f"{passenger}'s flight has been rebooked from {old_flight} to {new_flight}.",
        "success"
    )

    add_notification(
        "Updated Ticket Generated",
        f"New PNR {pnr} has been generated for {new_flight}.",
        "ticket"
    )

    return jsonify({
        "success": True,
        "ticket": ticket
    })

/* =========================================================
   CHECK FLIGHT
========================================================= */

async function checkFlight(){

    const passenger =
        document.getElementById("passenger").value.trim();

    const flight =
        document.getElementById("flight").value.trim().toUpperCase();

    if(!passenger){

        showToast(
            "Passenger Name Required",
            "Please enter passenger name."
        );

        return;
    }

    if(!flight){

        showToast(
            "Flight Number Required",
            "Please enter your flight number."
        );

        return;
    }

    currentPassenger = passenger;

    currentFlight = flight;

    resetAgents();

    document.getElementById("analysisText").innerText =
        "Multi-agent system is analyzing your flight...";

    const response = await fetch(
        "/api/analyze",
        {
            method:"POST",

            headers:{
                "Content-Type":
                "application/json"
            },

            body:JSON.stringify({
                passenger,
                flight
            })
        }
    );

    const data = await response.json();

    if(!data.success){

        showToast(
            "Flight Not Found",
            data.message
        );

        return;
    }


    await runAgents(data.flight);

    displayFlight(data.flight);

    loadNotifications();

}


/* =========================================================
   RESET AGENTS
========================================================= */

function resetAgents(){

    for(let i=0;i<7;i++){

        const agent =
            document.getElementById(
                "agent"+i
            );

        agent.classList.remove(
            "processing",
            "complete"
        );

        agent.querySelector(
            ".agent-state"
        ).innerText =
            "Waiting";
    }
}


/* =========================================================
   ONE-BY-ONE AGENT PROCESS
========================================================= */

async function runAgents(flight){

    const names = [

        "Verifying passenger booking...",

        "Tracking aircraft position...",

        "Analyzing weather conditions...",

        "Detecting flight disruption...",

        "Predicting operational risk...",

        "Finding best passenger decision...",

        "Preparing passenger notification..."

    ];


    for(let i=0;i<7;i++){

        const agent =
            document.getElementById(
                "agent"+i
            );

        agent.classList.add(
            "processing"
        );

        agent.querySelector(
            ".agent-state"
        ).innerText =
            "Processing...";

        document.getElementById(
            "analysisText"
        ).innerText =
            names[i];


        await sleep(700);


        agent.classList.remove(
            "processing"
        );

        agent.classList.add(
            "complete"
        );

        agent.querySelector(
            ".agent-state"
        ).innerText =
            "Complete";

    }


    document.getElementById(
        "analysisText"
    ).innerText =
        "Analysis completed. Passenger action is ready.";

}

async function loadSeats(flight){

    document.getElementById(
        "seatFlightName"
    ).innerText =
        `${flight.airline} | ${flight.flight} | ${flight.aircraft}`;


    document.getElementById(
        "seatFlightDetails"
    ).innerHTML = `
        ${flight.from} → ${flight.to}<br>
        ${flight.departure} → ${flight.arrival}<br>
        ${flight.terminal} |
        Gate ${flight.gate}
    `;


    const response =
        await fetch(
            "/api/seats/"
            + encodeURIComponent(
                flight.flight
            )
        );


    const data =
        await response.json();


    const rows =
        document.getElementById(
            "seatRows"
        );


    rows.innerHTML = "";


    for(
        let row=1;
        row<=data.rows;
        row++
    ){

        const rowDiv =
            document.createElement(
                "div"
            );

        rowDiv.className =
            "seat-row";


        const number =
            document.createElement(
                "div"
            );

        number.className =
            "row-number";

        number.innerText =
            row;


        rowDiv.appendChild(
            number
        );


        data.columns.forEach(
            col=>{

                if(col === "D"){

                    const aisle =
                        document.createElement(
                            "div"
                        );

                    aisle.style.width =
                        "10px";

                    rowDiv.appendChild(
                        aisle
                    );

                }


                const seat =
                    document.createElement(
                        "button"
                    );

                const seatId =
                    row + col;


                seat.className =
                    "seat";

                seat.innerText =
                    seatId;


                if(
                    data.occupied
                    .includes(seatId)
                ){

                    seat.classList.add(
                        "occupied"
                    );

                    seat.disabled =
                        true;

                }else{

                    seat.onclick =
                        function(){

                            chooseSeat(

                            );

                        };

        );


        rows.appendChild(
            rowDiv
        );

    }

/* =========================================================
   CONFIRM REBOOK
========================================================= */

async function confirmSeat(){

    if(!selectedAlternative){

        showToast(
            "Flight Required",
            "Please select an alternative flight first."
        );

        return;
    }


    if(!selectedSeat){

        showToast(
            "Seat Required",
            "Please select an available seat."
        );

        return;
    }


    const response =
        await fetch(
            "/api/rebook",
            {
                method:"POST",

                headers:{
                    "Content-Type":
                    "application/json"
                },

                body:JSON.stringify({

                    passenger:
                        currentPassenger,

                    old_flight:
                        currentFlight,

                    new_flight:
                        selectedAlternative.flight,

                    seat:
                        selectedSeat

                })
            }
        );


    const data =
        await response.json();


    if(!data.success){

        showToast(
            "Rebooking Failed",
            data.message
        );

        return;
    }


    closeSeatModal();

    showTicket(
        data.ticket
    );

    loadNotifications();

    loadBookings();

}

async function loadNotifications(){

function renderNotifications(
    list,
    target,
    limit
){

    const container =
        document.getElementById(
            target
        );


    if(!container)
        return;


    container.innerHTML = "";


    list.slice(0,limit)
        .forEach(
            item=>{

                let icon =
                    item.type === "success"
                    ? "✓"
                    : item.type === "alternative"
                    ? "✈"
                    : item.type === "ticket"
                    ? "🎫"
                    : "🔔";


                container.innerHTML += `

                    <div class="notification">

                        <div class="notification-icon">
                            ${icon}
                        </div>

                        <div>

                            <h4>
                                ${item.title}
                            </h4>

                            <p>
                                ${item.message}
                            </p>

                            <div class="notification-time">
                                ${item.time}
                            </div>

                        </div>

                    </div>

                `;

            }
        );
}
